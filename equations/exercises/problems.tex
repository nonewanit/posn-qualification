% --- Part 1: พื้นฐาน (Basic) ---
\examsection{ตอนที่ 1: พื้นฐาน — สมการเชิงเส้นและสมการกำลังสอง}

\problem{
  จงแก้สมการ $5x - 3 = 2x + 9$
}{
  $x = 4$
}

\problem{
  จงแก้สมการ $\dfrac{x}{4} + \dfrac{x}{3} = 7$
}{
  $x = 12$
}

\problem{
  จงแก้สมการ $x^2 - 7x + 12 = 0$ โดยการแยกตัวประกอบ
}{
  $(x - 3)(x - 4) = 0$,
  $x = 3$ หรือ $x = 4$
}

\problem{
  จงแก้สมการ $x^2 - 16 = 0$
}{
  $(x - 4)(x + 4) = 0$,
  $x = 4$ หรือ $x = -4$
}

\problem{
  จงแก้สมการ $2x^2 + 5x - 3 = 0$ โดยใช้สูตรกำลังสอง
}{
  $a = 2, b = 5, c = -3$

  $\Delta = 25 - 4(2)(-3) = 49$

  $x = \frac{-5 \pm 7}{4}$,
  $x = \frac{1}{2}$ หรือ $x = -3$
}

% --- Part 2: การประยุกต์ (Intermediate) ---
\examsection{ตอนที่ 2: การประยุกต์ — ระบบสมการและอสมการ}

\problem{
  จงแก้ระบบสมการ
  \[
    \begin{cases}
      3x + 2y = 8 \\
      x - y = 1
    \end{cases}
  \]
}{
  $3x + 2y = 8$, \; $x = y + 1$

  $3(y+1) + 2y = 8$,
  $3y + 3 + 2y = 8$,
  $5y = 5$,
  $y = 1$,
  $x = 2$

  $(x, y) = (2, 1)$
}

\problem{
  จงแก้อสมการ $4x - 7 < 2x + 5$ และเขียนคำตอบในรูปช่วง
}{
  $2x < 12$,
  $x < 6$,
  $x \in (-\infty, 6)$
}

\problem{
  จงแก้อสมการ $3 - x \geq 2x + 9$
}{
  $-3x \geq 6$,
  $x \leq -2$,
  $x \in (-\infty, -2]$
}

\problem{
  จงแก้อสมการ $x^2 - x - 6 \leq 0$ โดยใช้ Sign Chart
}{
  $(x - 3)(x + 2) \leq 0$, จุดวิกฤต: $x = -2, 3$

  \small
  \begin{tabular}{c|ccc}
    & $x<-2$ & $-2<x<3$ & $x>3$ \\
    \hline
    $(x-3)$ & $-$ & $-$ & $+$ \\
    $(x+2)$ & $-$ & $+$ & $+$ \\
    \hline
    ผลคูณ   & $+$ & $-$ & $+$
  \end{tabular}
  \normalsize

  $x \in [-2, 3]$
}

\problem{
  จงแก้สมการ $|2x + 3| = 7$
}{
  $2x+3 = 7 \Rightarrow 2x = 4 \Rightarrow x = 2$

  $2x+3 = -7 \Rightarrow 2x = -10 \Rightarrow x = -5$

  $x = 2$ หรือ $x = -5$
}

\problem{
  จงแก้อสมการ $|x - 1| < 4$ และเขียนคำตอบในรูปช่วง
}{
  $-4 < x - 1 < 4$,
  $-3 < x < 5$,
  $x \in (-3, 5)$
}

% --- Part 3: ระดับ สอวน. (POSN Level) ---
\examsection{ตอนที่ 3: ระดับข้อสอบ สอวน.}

\problem{
  จงหาค่าของ $k$ ที่ทำให้สมการ $x^2 + (k+1)x + (k+4) = 0$
  มีคำตอบเดียว (คำตอบซ้ำ)
}{
  มีคำตอบเดียวเมื่อ $\Delta = 0$:

  $(k+1)^2 - 4(1)(k+4) = 0$

  $k^2 + 2k + 1 - 4k - 16 = 0$

  $k^2 - 2k - 15 = 0$

  $(k-5)(k+3) = 0$

  $k = 5$ หรือ $k = -3$
}

\problem{
  จงแก้สมการ $x^3 - 7x^2 + 10x = 0$
}{
  $x(x^2 - 7x + 10) = 0$

  $x(x - 2)(x - 5) = 0$

  $x = 0$ หรือ $x = 2$ หรือ $x = 5$
}

\problem{
  จงแก้อสมการ $x^3 - 4x^2 \leq 0$ โดยใช้ Sign Chart
}{
  $x^2(x - 4) \leq 0$, จุดวิกฤต: $x = 0, 4$

  $x^2 \geq 0$ เสมอ ดังนั้นเครื่องหมายขึ้นกับ $(x-4)$:

  \small
  \begin{tabular}{c|ccc}
    & $x<0$ & $0<x<4$ & $x>4$ \\
    \hline
    $x^2$     & $+$ & $+$ & $+$ \\
    $(x-4)$   & $-$ & $-$ & $+$ \\
    \hline
    ผลคูณ     & $-$ & $-$ & $+$
  \end{tabular}
  \normalsize

  ต้องการ $\leq 0$: $x \in (-\infty, 4]$
}

\problem{
  จงแก้ระบบสมการ
  \[
    \begin{cases}
      x^2 + y = 7 \\
      x + y = 5
    \end{cases}
  \]
}{
  จากสมการที่ 2: $y = 5 - x$

  แทนในสมการที่ 1: $x^2 + (5 - x) = 7$

  $x^2 - x - 2 = 0$

  $(x - 2)(x + 1) = 0$

  $x = 2 \Rightarrow y = 3$ \quad หรือ \quad $x = -1 \Rightarrow y = 6$

  $(x, y) = (2, 3)$ หรือ $(-1, 6)$
}

\problem{
  จงหาค่าของ $m$ ที่ทำให้สมการ $mx^2 + (m-1)x + 1 = 0$
  มีคำตอบเป็นจำนวนจริง 1 คำตอบพอดี
}{
  มีคำตอบเดียวเมื่อ $\Delta = 0$:

  $(m-1)^2 - 4m = 0$

  $m^2 - 2m + 1 - 4m = 0$

  $m^2 - 6m + 1 = 0$

  $m = \frac{6 \pm \sqrt{36 - 4}}{2} = 3 \pm 2\sqrt{2}$
}

\problem{
  จงแก้สมการ $|x^2 - 4| = 5$
}{
  $x^2 - 4 = 5 \;\Rightarrow\; x^2 = 9 \;\Rightarrow\; x = \pm 3$

  $x^2 - 4 = -5 \;\Rightarrow\; x^2 = -1$ \;(ไม่มีคำตอบใน $\R$)

  $x = 3$ หรือ $x = -3$
}

\problem{
  จงแก้สมการ $\dfrac{2}{x-1} + \dfrac{1}{x+1} = \dfrac{3}{x^2-1}$
  (ระวัง: ระบุค่า $x$ ที่ไม่สามารถใช้ได้ด้วย)
}{
  $x \neq 1, -1$

  $\frac{2(x+1) + (x-1)}{(x-1)(x+1)} = \frac{3}{x^2-1}$

  $\frac{2x+2 + x-1}{x^2-1} = \frac{3}{x^2-1}$

  $\frac{3x+1}{x^2-1} = \frac{3}{x^2-1}$

  $3x + 1 = 3$,
  $3x = 2$,
  $x = \frac{2}{3}$

  (ตรวจสอบ: $x \neq \pm 1$, ใช้ได้ $\checkmark$)
}
